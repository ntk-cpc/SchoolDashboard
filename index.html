function doGet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getActiveSheet(); // ============================================================ // 讀取所有資料 // ============================================================ var lastRow = sheet.getLastRow(); var lastCol = sheet.getLastColumn(); if (lastRow < 2) { return createEmptyResponse(); } var allData = sheet.getRange(1, 1, lastRow, lastCol).getValues(); // ============================================================ // 第 1 行：頂部資料（根據你張圖嘅實際位置） // ============================================================ var header = allData[0]; // 根據你張圖嘅實際排列： // A1: "請輸入/檢視日期" (標籤) // B1: 日期值 (2026/9/3) // C1: "當天星期：" (標籤) // D1: 星期值 (星期四) // E1: "學校活動 / 備註：" (標籤) // F1: 活動值 (空白) // G1: "當天 Cycle：" (標籤) // H1: Cycle 值 (Day 1) // I1: "課節模式：" (標籤) // J1: 模式值 (全日) var dateVal = header[1] || ""; var weekdayVal = header[3] || ""; var noteVal = header[5] || ""; var cycleVal = header[7] || ""; var modeVal = header[9] || ""; // 格式化日期 var formattedDate = ""; if (dateVal) { try { var d = new Date(dateVal); if (!isNaN(d.getTime())) { formattedDate = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy/MM/dd"); } else { formattedDate = dateVal.toString().trim(); } } catch (e) { formattedDate = dateVal.toString().trim(); } } // 格式化星期 var formattedWeekday = ""; if (weekdayVal) { try { var wd = new Date(weekdayVal); if (!isNaN(wd.getTime())) { var weekdays = ["星期日", "星期一", "星期二", "星期三", "星期四", "星期五", "星期六"]; formattedWeekday = weekdays[wd.getDay()]; } else { formattedWeekday = weekdayVal.toString().trim(); } } catch (e) { formattedWeekday = weekdayVal.toString().trim(); } } // ============================================================ // 讀取課表（由第 2 行開始） // 根據你張圖：第1欄=課節，第2欄=時間，第3欄=課堂資訊 // ============================================================ var schedule = []; for (var r = 2; r < allData.length; r++) { var row = allData[r]; // 如果呢行係空白或標題行，跳過 var period = row[0] ? row[0].toString().trim() : ""; var time = row[1] ? row[1].toString().trim() : ""; var info = row[2] ? row[2].toString().trim() : ""; // 跳過空行同標題行 if (!period && !time && !info) continue; if (period === "課節" || period === "節次") continue; if (period === "截止日期" || period === "開始時間") break; // 去到其他 section 就停 schedule.push({ period: period, time: time, info: info }); } // ============================================================ // 回傳 JSON // ============================================================ var result = { date: formattedDate, weekday: formattedWeekday, note: noteVal ? noteVal.toString().trim() : "", cycleDay: cycleVal ? cycleVal.toString().trim() : "", mode: modeVal ? modeVal.toString().trim() : "全日", schedule: schedule, activities: [], homework: [], deadlines: [], meetings: [] }; return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function createEmptyResponse() { return ContentService .createTextOutput(JSON.stringify({ date: "", weekday: "", note: "", cycleDay: "", mode: "全日", schedule: [], activities: [], homework: [], deadlines: [], meetings: [] })) .setMimeType(ContentService.MimeType.JSON); }

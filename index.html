async function fetchDashboardData() { try { const response = await fetch(GOOGLE_API_URL, { redirect: "follow" }); const data = await response.json(); console.log("📊 接收到嘅資料:", data); document.getElementById("current-date-display").textContent = data.date + " (" + data.weekday + ")"; document.getElementById("cycle-day").textContent = data.cycleDay || "--"; document.getElementById("school-mode").textContent = data.mode || "全日"; document.getElementById("school-note").textContent = data.note || "無特別活動"; // ---- 課表 ---- const tbody = document.getElementById("schedule-body"); tbody.innerHTML = ""; const scheduleData = data.schedule || []; if (scheduleData.length > 0) { const now = new Date(); const currentMin = now.getHours() * 60 + now.getMinutes(); scheduleData.forEach(item => { const tr = document.createElement("tr"); let extraClass = ""; let isNow = false; if (item.info && item.info.includes("小息")) extraClass = "break"; else if (item.info && (item.info.includes("午膳") || item.info.includes("午睡"))) extraClass = "lunch"; if (item.time && item.time.includes('-')) { const [start, end] = item.time.split('-'); const [sh, sm] = start.split(':').map(Number); const [eh, em] = end.split(':').map(Number); if (!isNaN(sh) && !isNaN(sm) && !isNaN(eh) && !isNaN(em)) { const startMin = sh * 60 + sm; const endMin = eh * 60 + em; if (currentMin >= startMin && currentMin < endMin) isNow = true; } } if (isNow) extraClass += " now"; if (extraClass) tr.className = extraClass.trim(); tr.innerHTML = ` ${item.period || ''} ${item.time || ''} ${item.info || ''} `; tbody.appendChild(tr); }); } else { tbody.innerHTML = `今日無課堂`; } // ---- 學生活動（已改顯示順序） ---- const actBox = document.getElementById("activity-container"); actBox.innerHTML = ""; if (data.activities && data.activities.length > 0) { data.activities.forEach(a => { const div = document.createElement("div"); div.className = "list-item activity"; // ✅ 改咗呢度：活動名稱 + 地點 + 時間 div.innerHTML = ` ${a.title} ${a.location ? '(@' + a.location + ')' : ''} ${a.start || ''}-${a.end || ''} `; actBox.appendChild(div); }); } else { actBox.innerHTML = `
今日無學生活動
`; } // ---- 功課 ---- const hwBox = document.getElementById("homework-container"); hwBox.innerHTML = ""; if (data.homework && data.homework.length > 0) { data.homework.forEach(h => { const div = document.createElement("div"); div.className = "list-item hw"; div.innerHTML = ` [${h.type}] ${h.className} - ${h.title} `; hwBox.appendChild(div); }); } else { hwBox.innerHTML = `
今日無功課紀錄
`; } // ---- Deadline ---- const dlBox = document.getElementById("deadline-container"); dlBox.innerHTML = ""; if (data.deadlines && data.deadlines.length > 0) { data.deadlines.forEach(d => { const div = document.createElement("div"); div.className = "list-item dl"; div.innerHTML = ` ${d.item} (${d.dept}) ${d.status} `; dlBox.appendChild(div); }); } else { dlBox.innerHTML = `
今日無截止需求
`; } // ---- 會議 ---- const mBox = document.getElementById("meetings-container"); mBox.innerHTML = ""; if (data.meetings && data.meetings.length > 0) { data.meetings.forEach(m => { const div = document.createElement("div"); div.className = "list-item mtg"; div.innerHTML = ` ${m.title} ${m.location || ''} ${m.start || ''}-${m.end || ''} `; mBox.appendChild(div); }); } else { mBox.innerHTML = `
今日無會議
"; } } catch (err) { console.error("Google Sheets 同步失敗:", err); } }

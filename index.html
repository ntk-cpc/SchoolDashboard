// ============================================ // Web App API - 提供 JSON 資料 // ============================================ function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("每日主控台"); if (!sheet) { return createErrorResponse("找不到『每日主控台』工作表"); } // ===== 取得參數 ===== var params = e.parameter || {}; var action = params.action || "dashboard"; // dashboard, homework, timetable, activities, meetings, files, all // ===== 根據 action 回傳不同資料 ===== var result = {}; switch(action) { case "dashboard": result = getDashboardData(sheet); break; case "homework": result = getHomeworkData(sheet); break; case "timetable": result = getTimetableData(sheet); break; case "activities": result = getActivitiesData(sheet); break; case "meetings": result = getMeetingsData(sheet); break; case "files": result = getFilesData(sheet); break; case "all": default: result = getAllData(sheet); break; } // ===== 加入狀態資訊 ===== result.status = "success"; result.timestamp = new Date().toISOString(); result.timezone = "Asia/Hong_Kong"; return ContentService.createTextOutput(JSON.stringify(result, null, 2)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return createErrorResponse("錯誤：" + error.toString()); } } // ============================================ // 各資料擷取函數 // ============================================ /** * 取得儀表板主要資料（日期、Cycle、模式等） */ function getDashboardData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "dashboard", date: "", time: "", cycle: "", mode: "", activities: [], shuttleBus: [] }; // 找日期時間（通常在 C1 或 D1） for (var i = 0; i < data.length && i < 5; i++) { var row = data[i]; for (var j = 0; j < row.length; j++) { var cell = String(row[j] || ""); // 找日期格式 if (cell.match(/\d{2}\/\d{2}\/\d{4}/)) { result.date = cell; } // 找時間格式 if (cell.match(/\d{2}:\d{2}:\d{2}/)) { result.time = cell; } // 找 Cycle if (cell.match(/Day\s*\d/)) { result.cycle = cell; } // 找模式 if (cell === "全日" || cell === "上午" || cell === "下午") { result.mode = cell; } } } // 擷取學生活動（找到「開始時間」欄位之後的資料） var activities = extractSection(data, "開始時間", "學生活動內容", ["開始時間", "結束時間", "學生活動內容", "地點"]); result.activities = activities; // 擷取穿梭巴士 var shuttle = extractSection(data, "屋苑穿梭巴士", "由", ["路線", "時間"]); result.shuttleBus = shuttle; return result; } /** * 取得今日課表 */ function getTimetableData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "timetable", date: "", dayOfWeek: "", schedule: [] }; // 找日期 for (var i = 0; i < data.length && i < 3; i++) { var row = data[i]; for (var j = 0; j < row.length; j++) { var cell = String(row[j] || ""); if (cell.match(/\d{2}\/\d{2}\/\d{4}/)) { result.date = cell; } if (cell.match(/星期[一二三四五六日]/)) { result.dayOfWeek = cell; } } } // 擷取課表（找到「課節」欄位之後的資料） var timetable = extractSection(data, "課節", "課堂資訊", ["課節", "時間", "課堂資訊"]); result.schedule = timetable; return result; } /** * 取得學生活動 */ function getActivitiesData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "activities", activities: extractSection(data, "開始時間", "學生活動內容", ["開始時間", "結束時間", "學生活動內容", "地點"]) }; return result; } /** * 取得會議 */ function getMeetingsData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "meetings", meetings: extractSection(data, "日期", "會議內容", ["日期", "開始時間", "結束時間", "會議內容", "地點"]) }; return result; } /** * 取得功課（從功課總表） */ function getHomeworkData(sheet) { // 如果「每日主控台」有功課資料，直接讀取 var data = sheet.getDataRange().getValues(); var result = { type: "homework", homework: extractSection(data, "類別", "功課項目", ["類別", "班別", "功課項目"]) }; // 如果沒有，嘗試從「功課總表」讀取 if (result.homework.length === 0) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var homeworkSheet = ss.getSheetByName("功課總表"); if (homeworkSheet) { var hwData = homeworkSheet.getDataRange().getValues(); var today = Utilities.formatDate(new Date(), "Asia/Hong_Kong", "yyyy/M/d"); var hwResults = []; for (var i = 1; i < hwData.length; i++) { var row = hwData[i]; var rowDate = formatDate(row[0]); if (rowDate === today) { hwResults.push({ category: row[1] || "", class: row[2] || "", item: row[3] || "" }); } } result.homework = hwResults; result.source = "功課總表"; } } catch(e) { // 忽略錯誤 } } return result; } /** * 取得文件 Deadline */ function getFilesData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "files", files: extractSection(data, "文件 / 教務項目", "完成狀態", ["文件 / 教務項目", "負責部門", "完成狀態"]) }; return result; } /** * 取得所有資料（完整輸出） */ function getAllData(sheet) { var data = sheet.getDataRange().getValues(); var result = { type: "all", dashboard: getDashboardData(sheet), timetable: getTimetableData(sheet), activities: getActivitiesData(sheet), meetings: getMeetingsData(sheet), files: getFilesData(sheet), homework: getHomeworkData(sheet), rawData: { headers: data[0] || [], rowCount: data.length, colCount: data[0] ? data[0].length : 0 } }; return result; } // ============================================ // 輔助函數 // ============================================ /** * 從試算表中擷取特定區塊的資料 */ function extractSection(data, startKeyword, endKeyword, columnNames) { var results = []; var startRow = -1; var endRow = -1; var colMap = {}; // 尋找開始行 for (var i = 0; i < data.length; i++) { var row = data[i]; if (!row) continue; for (var j = 0; j < row.length; j++) { var cell = String(row[j] || ""); if (cell.includes(startKeyword)) { startRow = i; break; } } if (startRow !== -1) break; } if (startRow === -1) return results; // 建立欄位對應 var headerRow = data[startRow]; for (var j = 0; j < headerRow.length; j++) { var header = String(headerRow[j] || ""); for (var k = 0; k < columnNames.length; k++) { if (header.includes(columnNames[k])) { colMap[columnNames[k]] = j; break; } } } // 檢查是否找到所有欄位 var foundAll = true; for (var k = 0; k < columnNames.length; k++) { if (!(columnNames[k] in colMap)) { foundAll = false; break; } } if (!foundAll) return results; // 讀取資料行（從 startRow + 1 開始） for (var i = startRow + 1; i < data.length; i++) { var row = data[i]; if (!row || !row[0]) break; // 檢查是否到達結束行 var isEnd = false; for (var j = 0; j < row.length; j++) { var cell = String(row[j] || ""); if (cell.includes(endKeyword) && i > startRow + 1) { isEnd = true; break; } } if (isEnd) break; // 建立結果物件 var entry = {}; for (var k = 0; k < columnNames.length; k++) { var colName = columnNames[k]; var colIndex = colMap[colName]; entry[colName] = (colIndex !== undefined && row[colIndex] !== undefined) ? row[colIndex] : ""; } results.push(entry); } return results; } /** * 格式化日期 */ function formatDate(date) { if (!date) return ""; if (date instanceof Date) { return Utilities.formatDate(date, "Asia/Hong_Kong", "yyyy/M/d"); } if (typeof date === "string") { var temp = new Date(date); if (!isNaN(temp)) { return Utilities.formatDate(temp, "Asia/Hong_Kong", "yyyy/M/d"); } return date; } return String(date); } /** * 建立錯誤回應 */ function createErrorResponse(message) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: message, timestamp: new Date().toISOString() }, null, 2)).setMimeType(ContentService.MimeType.JSON); }
